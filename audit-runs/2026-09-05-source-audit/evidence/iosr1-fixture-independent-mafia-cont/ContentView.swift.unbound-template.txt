import SwiftUI
import Foundation
#error("Audit template: verify the generated Kotlin header, bind both call tokens, then remove this line")
import UIKit
import ComposeApp

/// Hosts the Kotlin Compose Multiplatform root view inside a SwiftUI scene.
///
/// `MainViewController()` is exported from `:composeApp` (see
/// `composeApp/src/iosMain/.../MainViewController.kt`). It builds a
/// `UIViewController` that wraps the shared `App()` composable and
/// starts Koin on first call.
struct ContentView: View {
    @Environment(\.scenePhase) private var scenePhase
    @State private var auditStatus = "idle"

    private var auditEnabled: Bool {
        ProcessInfo.processInfo.arguments.contains("--parlor-audit-iosr1")
    }

    var body: some View {
        ZStack {
            ComposeView()
                .frame(maxWidth: .infinity, maxHeight: .infinity)
                .accessibilityHidden(scenePhase != .active)

            // Copy-only diagnostic trigger. The existing Compose init remains unchanged.
            // Keep this below the original privacy cover and absent while inactive.
            if auditEnabled && scenePhase == .active {
                VStack(alignment: .trailing) {
                    Spacer()
                    Button("Run recovery audit") { runRecoveryAudit() }
                        .accessibilityIdentifier("parlor-audit-iosr1-trigger")
                        .disabled(auditStatus != "idle")
                    Text(auditStatus)
                        .accessibilityIdentifier(auditStatus == "complete"
                            ? "parlor-audit-iosr1-complete" : "parlor-audit-iosr1-state")
                }
                .frame(maxWidth: .infinity, maxHeight: .infinity, alignment: .bottomTrailing)
                .padding(24)
            }

            // iOS snapshots the scene for the app switcher after it becomes
            // inactive. Cover roles and private actions before that snapshot,
            // but do not suspend the LAN session until true background.
            if scenePhase != .active {
                Color.black
                    .accessibilityHidden(true)
            }
        }
        .frame(maxWidth: .infinity, maxHeight: .infinity)
        .ignoresSafeArea(.container, edges: .all)
        .onAppear {
            reportScenePhase(scenePhase)
        }
        .onChange(of: scenePhase) { phase in
            reportScenePhase(phase)
        }
        .onDisappear {
            if auditEnabled { __PARLOR_AUDIT_IOSR1_CANCEL_CALL__() }
        }
    }

    private func runRecoveryAudit() {
        guard auditEnabled && auditStatus == "idle" else { return }
        auditStatus = "running"
        // Replace token only after inspecting this build's generated Kotlin header.
        __PARLOR_AUDIT_IOSR1_START_CALL__ { json in
            guard json.utf8.count <= 8192 else {
                auditStatus = "result_too_large"
                return
            }
            let output = URL(fileURLWithPath: NSTemporaryDirectory(), isDirectory: true)
                .appendingPathComponent("parlor-audit-iosr1-result.json", isDirectory: false)
            guard !FileManager.default.fileExists(atPath: output.path) else {
                auditStatus = "existing_output"
                return
            }
            do {
                try json.write(to: output, atomically: true, encoding: .utf8)
                // This means evidence was written, NOT that recovery is healthy.
                auditStatus = "complete"
            } catch {
                auditStatus = "output_failed" // Do not print NSError descriptions/paths.
            }
        }
    }

    private func reportScenePhase(_ phase: ScenePhase) {
        switch phase {
        case .active:
            MainViewControllerKt.NotifyAppForegrounded()
        case .inactive:
            MainViewControllerKt.NotifyAppInactive()
        case .background:
            MainViewControllerKt.NotifyAppBackgrounded()
        @unknown default:
            // An unknown state is privacy-covered but is not evidence that the
            // process entered background, so keep transport state unchanged.
            MainViewControllerKt.NotifyAppInactive()
        }
    }
}

private struct ComposeView: UIViewControllerRepresentable {
    func makeUIViewController(context: Context) -> UIViewController {
        MainViewControllerKt.MainViewController()
    }

    func updateUIViewController(_ uiViewController: UIViewController, context: Context) {
        // No-op — the Kotlin view controller owns its own state.
    }
}
