25:         namespace: IdentityNamespace,
26:         fingerprintDigest: (EncodedIdentityKeyPair) -> ByteArray,
27:         generate: () -> EncodedIdentityKeyPair
28:     ): EncodedIdentityKeyPair {
29:         rejectPendingReset(namespace)
30:         val identityKey = identityKey(namespace)
31:         val existing = read(identityKey)
32:         if (existing != null) {
33:             return validateAndReturn(decodeAndClear(namespace, existing), fingerprintDigest)
34:         }
35: 
36:         val candidate = generate()
37:         val candidateRecord = IdentityKeyRecordCodec.encode(namespace, candidate)
38:         val storeInput = candidateRecord.copyOf()
39:         var winner: ByteArray? = null
40:         var reread: ByteArray? = null
41:         try {
42:             val durableWinner = callStore("atomically persist secure identity") {
43:                 store.putIfAbsent(identityKey, storeInput)
44:             }
45:             winner = durableWinner
46:             if (durableWinner.size != IdentityKeyRecordCodec.RECORD_SIZE) {
47:                 throw corrupt("JVM secure store returned a non-P2KI winner")
48:             }
49:             val durableReread = read(identityKey)
50:                 ?: contractViolation("durable winner was absent on immediate reread")
51:             reread = durableReread
52:             if (!constantTimeEquals(durableWinner, durableReread)) {
53:                 contractViolation("putIfAbsent winner differed from immediate reread")
54:             }
55:             return validateAndReturn(
56:                 IdentityKeyRecordCodec.decode(namespace, durableReread),
57:                 fingerprintDigest
58:             )
59:         } catch (e: IdentityRecordCorruptException) {
60:             throw corrupt(e.message ?: "JVM secure identity record is corrupt", e)
61:         } finally {
62:             candidate.clearPrivate()
63:             candidateRecord.fill(0)
64:             storeInput.fill(0)
65:             winner?.fill(0)
66:             reread?.fill(0)
67:         }

227: /**
228:  * Process-local reset/kit exclusion for one exact host-store instance.
229:  * Cross-process exclusion remains part of [JvmSecureIdentityStore]'s host
230:  * contract and cannot be inferred by core.
231:  */
232: private object JvmSecureIdentityLiveGuard {
233:     private val lock = Any()
234:     private val states = mutableMapOf<StoreNamespace, State>()
235: 
236:     fun acquire(store: JvmSecureIdentityStore, namespace: String): SecureIdentityUsage {
237:         val key = StoreNamespace(store, namespace)
238:         synchronized(lock) {
239:             val state = states.getOrPut(key) { State() }
240:             if (state.resetInProgress) {
241:                 throw localIdentityError(
242:                     kind = LocalIdentityFailureKind.TEMPORARILY_UNAVAILABLE,
243:                     recovery = LocalIdentityRecovery.RETRY,
244:                     reason = "JVM secure identity reset is already in progress"
245:                 )
246:             }
247:             state.liveUsages++
248:         }
249: 
250:         val released = AtomicBoolean(false)
251:         return SecureIdentityUsage {
252:             if (!released.compareAndSet(false, true)) return@SecureIdentityUsage
253:             synchronized(lock) {
254:                 val state = checkNotNull(states[key]) {
255:                     "JVM secure identity usage state disappeared before release"
256:                 }
257:                 check(state.liveUsages > 0) { "JVM secure identity usage count underflow" }
258:                 state.liveUsages--
259:                 if (state.liveUsages == 0 && !state.resetInProgress) states.remove(key)
260:             }
261:         }
262:     }
300:     }
301: 
302:     private class State(
303:         var liveUsages: Int = 0,
304:         var resetInProgress: Boolean = false
305:     )
306: 
307:     private class StoreNamespace(
308:         private val store: JvmSecureIdentityStore,
309:         private val namespace: String
310:     ) {
311:         override fun equals(other: Any?): Boolean =
312:             other is StoreNamespace && store === other.store && namespace == other.namespace
313: 
314:         override fun hashCode(): Int = 31 * System.identityHashCode(store) + namespace.hashCode()
315:     }
316: }

13:     private val namespaceLabel = "dev.p2pkit.identity-namespace.v2\u0000".encodeToByteArray()
14:     private val fingerprintLabel = "dev.p2pkit.x25519-fingerprint.v1\u0000".encodeToByteArray()
15:     private val peerIdLabel = "dev.p2pkit.peer-id.v2\u0000".encodeToByteArray()
16:     private val appBindingLabel = "dev.p2pkit.app-binding.v1\u0000".encodeToByteArray()
17: 
18:     fun namespace(appId: AppId, cryptography: IdentityCryptography): IdentityNamespace {
19:         if (!appId.value.isWellFormedUtf16()) {
20:             throw P2pError.SecurityConfigurationInvalid(
21:                 "Secure v2 AppId must contain well-formed Unicode"
22:             )
23:         }
24:         val appIdBytes = appId.value.encodeToByteArray()
25:         if (appIdBytes.size > UShort.MAX_VALUE.toInt()) {
26:             throw P2pError.SecurityConfigurationInvalid(
27:                 "Secure v2 AppId UTF-8 encoding must fit an unsigned 16-bit length"
28:             )
29:         }
30:         val hash = sha256(
31:             cryptography,
32:             concatenate(namespaceLabel, u16be(appIdBytes.size), appIdBytes)
33:         )
34:         return IdentityNamespace(appId, appIdBytes, hash)
35:     }
36: 
37:     fun fingerprintDigest(
38:         publicKey: ByteArray,
39:         cryptography: IdentityCryptography
40:     ): ByteArray {
41:         require(publicKey.size == X25519_KEY_SIZE_BYTES) { "X25519 public key must be 32 bytes" }
42:         return sha256(cryptography, concatenate(fingerprintLabel, publicKey))
43:     }
