import XCTest

final class IOSAppLaunchUITests: XCTestCase {
    override func setUpWithError() throws {
        continueAfterFailure = false
    }

    @MainActor
    func testAuditProductionRecoverySourceAttribution() throws {
        let app = XCUIApplication()
        app.launchArguments += ["-AppleLanguages", "(en)", "-AppleLocale", "en_US", "--parlor-audit-iosr1"]
        // Optional: append "--parlor-audit-iosr1-native" for a subsequent read only.
        app.launch()
        defer { app.terminate() }

        XCTAssertTrue(
            app.wait(for: .runningForeground, timeout: 30),
            "The Swift host should launch and remain in the foreground",
        )
        XCTAssertTrue(
            app.staticTexts["parlor-home-brand"].waitForExistence(timeout: 30),
            "The Kotlin Compose root should render the home screen",
        )
        XCTAssertFalse(
            app.alerts.firstMatch.waitForExistence(timeout: 2),
            "No alert should appear during the simulator cold-start observation window",
        )
        XCTAssertEqual(
            app.state,
            .runningForeground,
            "The Swift host should remain in the foreground after Compose renders",
        )

        let warningPresent = app.staticTexts["Some recovery information is unavailable."]
            .waitForExistence(timeout: 10)
        let initial = XCTAttachment(string: "original_home_warning_present=\(warningPresent)")
        initial.name = "Original Home observation before diagnostic rerun"
        initial.lifetime = .keepAlways
        add(initial)
        let image = XCTAttachment(screenshot: app.screenshot())
        image.name = "Original Home before diagnostic rerun (copy-only audit overlay present)"
        image.lifetime = .keepAlways
        add(image)

        let trigger = app.buttons["parlor-audit-iosr1-trigger"]
        XCTAssertTrue(trigger.waitForExistence(timeout: 10), "Missing copy-only audit trigger")
        trigger.tap()
        XCTAssertTrue(
            app.staticTexts["parlor-audit-iosr1-complete"].waitForExistence(timeout: 30),
            "No sanitized audit receipt; do not infer a production storage result",
        )
        // PASS here means original Home rendered and the diagnostic wrote a receipt.
        // Parent must parse harness_status and source outcomes; no health assertion is made.
    }
}
