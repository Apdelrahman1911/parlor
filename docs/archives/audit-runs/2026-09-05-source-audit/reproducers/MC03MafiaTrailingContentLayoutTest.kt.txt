/** Audit-only public shipping-composable witness; do not register in production source sets. */
@OptIn(ExperimentalTestApi::class)
class MC03MafiaTrailingContentLayoutTest {
    @Test
    fun validLongNameMustNotEraseItsFinalRoleAtCompactWidth() = runComposeUiTest {
        val longName = "W".repeat(RoomInputPolicy.MAX_DISPLAY_NAME_LENGTH)
        assertTrue(RoomInputPolicy.isValidDisplayName(longName))
        setContent {
            CompositionLocalProvider(LocalDensity provides Density(1f, fontScale = 1f)) {
                ParlorTheme(reducedMotion = true) {
                    PostGameScreen(
                        winner = Team.Town,
                        finalRoles = listOf(
                            longName to Role.Mafia,
                            "Audit B" to Role.Doctor,
                            "Audit C" to Role.Detective,
                            "Audit D" to Role.Civilian,
                            "Audit E" to Role.Civilian,
                        ),
                        onExit = {},
                        modifier = Modifier.size(width = 320.dp, height = 640.dp),
                    )
                }
            }
        }
        // The exact role spelling depends on active English/Arabic resource environment.
        val roleBounds = onNode(hasText("Mafia") or hasText("حرامي"), useUnmergedTree = true)
            .getUnclippedBoundsInRoot()
        val nameBounds = onNodeWithText(longName, useUnmergedTree = true).getUnclippedBoundsInRoot()
        assertTrue(
            roleBounds.width > 0.dp,
            "Final role lost all measured width: role=$roleBounds, name=$nameBounds",
        )
    }
}
