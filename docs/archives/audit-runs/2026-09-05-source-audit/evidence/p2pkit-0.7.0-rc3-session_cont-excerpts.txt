  947     override fun notifyAppBackgrounded() {
  948         when (backgroundPolicy) {
  949             is BackgroundPolicy.CloseActiveSessions -> {
  950                 // Close sessions + pause advertising/discovery, but do NOT emit
  951                 // P2pState.Stopped: the data transports stay bound (only stop()
  952                 // closes them) and the kit is still functional, so reporting
  953                 // Stopped would lie to host UIs and never recover on foreground.
  954                 // applyBackgroundPolicy already closes active sessions; don't
  955                 // double-close here.
  956                 scope.launch {
  957                     runCatching { stopAdvertising() }
  958                     runCatching { stopDiscovery() }
  959                 }
  960                 sessionManager.applyBackgroundPolicy(backgroundPolicy)
  961             }
  962             is BackgroundPolicy.KeepRunning -> { /* nothing to do */ }
  963         }
  964     }
